# PathFinder is public and indexable. The robots meta tag that used to sit in
# index.html is gone; this file is the protocol-level counterpart.
#
# Both this file and /sitemap.xml must exist as real files under public/. The
# nginx SPA history fallback (try_files ... /index.html) answers anything else
# with HTML, so a missing sitemap would be served as an HTML document.
#
# /ergebnis is excluded on purpose: it renders from state held in the visitor's
# own browser, so a crawler fetching it would index an empty results page.
User-agent: *
Allow: /
Disallow: /ergebnis

Sitemap: https://pathfinder-berufetest.de/sitemap.xml
